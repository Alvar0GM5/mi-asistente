async function sendMessage() { const inputField = document.getElementById("userInput"); const messageText = inputField.value.trim(); if (!messageText && selectedFiles.length === 0) return; const activeChat = chats[currentChatId]; if (activeChat.messages.length <= 1) { activeChat.title = (messageText || selectedFiles[0]?.name || "Conversación").slice(0, 25) + "..."; } let fileUris = []; const displayFileText = selectedFiles.length > 0 ? '\n\n*(Archivos adjuntos: ' + selectedFiles.map(f => f.name).join(', ') + ')*' : ''; activeChat.messages.push({ role: 'user', text: messageText + displayFileText }); saveState(); renderSidebar(); renderMessages(); inputField.value = ""; const filesToUpload = [...selectedFiles]; selectedFiles = []; renderFilePreview(); currentAbortController = new AbortController(); setSendButtonMode(true); try { if (filesToUpload.length > 0) { for (const file of filesToUpload) { const uploaded = await uploadFileToGemini(file); fileUris.push(uploaded); } } const response = await fetch("/api/chat", { method: "POST", headers: { "Content-Type": "application/json" }, signal: currentAbortController.signal, body: JSON.stringify({ message: messageText, fileUris: fileUris }) }); const data = await response.json(); if (!response.ok) { throw new Error(data.error || "Error al conectar con la API"); } activeChat.messages.push({ role: 'model', text: data.reply }); renderMessages(); saveState(); } catch (error) { if (error.name !== 'AbortError') { activeChat.messages.push({ role: 'model', text: 'Error: ' + error.message }); renderMessages(); saveState(); } } setSendButtonMode(false); currentAbortController = null; }
